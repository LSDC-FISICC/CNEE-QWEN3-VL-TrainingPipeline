## fusionar pesos LoRA con el modelo

cd ~/CNEE
source .venv/bin/activate

python3 << 'EOF'
from unsloth import FastVisionModel

print("Cargando modelo entrenado...")
model, tokenizer = FastVisionModel.from_pretrained(
    model_name="/home/julioefajardo/CNEE/output/qwen3vl_2b_cnee/final",
    load_in_4bit=True,
)

print("Fusionando pesos LoRA con modelo base...")
model.save_pretrained_merged(
    "/home/julioefajardo/CNEE/output/qwen3vl_2b_cnee/merged",
    tokenizer,
    save_method="merged_16bit",
)
print("Modelo fusionado guardado en: output/qwen3vl_2b_cnee/merged")
EOF

## Convertir a formato GGUF para Ollama

cd ~/CNEE
source .venv/bin/activate

python3 << 'EOF'
from unsloth import FastVisionModel

model, tokenizer = FastVisionModel.from_pretrained(
    model_name="/home/julioefajardo/CNEE/output/qwen3vl_2b_cnee/merged",
    load_in_4bit=False,
)

print("Convirtiendo a GGUF Q4_K_M...")
model.save_pretrained_gguf(
    "/home/julioefajardo/CNEE/output/qwen3vl_2b_cnee/gguf",
    tokenizer,
    quantization_method="q4_k_m",
)
print("Conversion completa")
EOF

## Crear modelfile para ollama

cat > /home/julioefajardo/CNEE/output/qwen3vl_2b_cnee/Modelfile << 'EOF'
FROM /home/julioefajardo/CNEE/output/qwen3vl_2b_cnee/merged_gguf/merged.Q4_K_M.gguf

SYSTEM """Eres un evaluador tecnico experto en analisis de expedientes de interrupcion electrica. Evaluas documentos segun 7 criterios normativos y produces una decision estructurada en JSON."""

PARAMETER temperature 0.1
PARAMETER top_p 0.9
PARAMETER num_ctx 8192
EOF

echo "Modelfile creado"
cat ~/CNEE/output/qwen3vl_2b_cnee/Modelfile
